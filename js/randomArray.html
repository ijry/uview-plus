# randomArray(array)

该函数可以打乱一维数组元素的顺序，这是随机过程

  • array <Array> 一维数组
import { randomArray } from '@/uni_modules/uview-plus';
import { onLoad } from '@dcloudio/uni-app';
import { ref } from 'vue';

const array = ref([1, 2, 3, 4, 5]);

onLoad(() => {
  console.log(randomArray(array.value));
});

import { randomArray } from '@/uni_modules/uview-plus';
export default{
  data() {
    return {
      array: [1, 2, 3, 4, 5]
    }
  },
  onLoad() {
    console.log(randomArray(this.array));
  }
}
